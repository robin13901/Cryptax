<task id="1" type="auto">
  <name>Task 1: Schema changes + migration</name>
  <files>
    packages/backend/src/db/schema.ts
    packages/backend/drizzle/0001_import_batches.sql
    packages/backend/drizzle/meta/_journal.json
  </files>
  <action>
    1. Add `importBatches` table to `packages/backend/src/db/schema.ts`:
       ```typescript
       export const importBatches = sqliteTable('import_batches', {
         id: integer('id').primaryKey({ autoIncrement: true }),
         filename: text('filename').notNull(),
         sourceType: text('source_type').notNull(),
         totalRows: integer('total_rows').notNull(),
         importedCount: integer('imported_count').notNull(),
         duplicatesCount: integer('duplicates_count').notNull(),
         errorsCount: integer('errors_count').notNull(),
         importedAt: text('imported_at').notNull(),
       });
       ```
    2. Add `batchId` column to the existing `transactions` table definition:
       ```typescript
       batchId: integer('batch_id').references(() => importBatches.id, { onDelete: 'set null' }),
       ```
       Place it after `importedAt`. This column is nullable (no `.notNull()`).
    3. Add an index for batch_id in the transactions table's third argument array:
       ```typescript
       index('idx_transactions_batch_id').on(t.batchId),
       ```
    4. Run `npx drizzle-kit generate` from the repo root to produce the migration SQL file.
    5. **CRITICAL**: After generation, manually edit the generated SQL file:
       - Add `STRICT` after the closing `)` of the `CREATE TABLE import_batches` statement (before the semicolon).
       - The ALTER TABLE for adding batch_id does NOT need STRICT (it modifies an existing STRICT table).
    6. Run `npx drizzle-kit migrate` to apply the migration.
    7. Verify the migration applied: `sqlite3 cryptax.db ".schema import_batches"` should show the table with STRICT.
    8. Run `npx biome format --write packages/backend/src/db/schema.ts` to fix formatting.
  </action>
  <verify>
    - `npx drizzle-kit migrate` completes without errors
    - The generated migration file exists in `packages/backend/drizzle/` and contains `STRICT`
    - `sqlite3 cryptax.db ".tables"` shows `import_batches`
    - `sqlite3 cryptax.db "PRAGMA table_info(transactions)"` shows `batch_id` column
  </verify>
  <done>import_batches table exists with STRICT; transactions has nullable batch_id FK; migration file committed</done>
</task>

<task id="2" type="auto">
  <name>Task 2: Install csv-parse + shared import types</name>
  <files>
    packages/backend/package.json
    packages/shared/src/types/import.ts
    packages/shared/src/types/index.ts
    packages/shared/src/index.ts
  </files>
  <action>
    1. Install csv-parse in the backend workspace:
       ```bash
       npm install csv-parse -w packages/backend
       ```
    2. Create `packages/shared/src/types/import.ts` with these types:
       ```typescript
       import type { SourceType } from './transaction.js';

       /** Represents a single import batch (one API call, possibly multiple files) */
       export interface ImportBatch {
         id: number;
         filename: string;
         sourceType: SourceType;
         totalRows: number;
         importedCount: number;
         duplicatesCount: number;
         errorsCount: number;
         importedAt: string;
       }

       /** Result of importing a single CSV file */
       export interface PerFileResult {
         filename: string;
         sourceType: SourceType;
         totalRows: number;
         imported: number;
         duplicatesSkipped: number;
         errors: ImportFileError[];
         batchId: number;
       }

       /** Error for a specific row in a CSV file */
       export interface ImportFileError {
         row: number;
         field: string;
         message: string;
         rawData?: string;
       }

       /** Aggregated summary across all files in one import request */
       export interface ImportResponse {
         results: PerFileResult[];
         summary: {
           totalFiles: number;
           totalRows: number;
           totalImported: number;
           totalDuplicates: number;
           totalErrors: number;
         };
       }
       ```
    3. Update `packages/shared/src/types/index.ts` to re-export the new types:
       ```typescript
       export type {
         ImportBatch,
         ImportFileError,
         ImportResponse,
         PerFileResult,
       } from './import.js';
       ```
       Add this block after the existing transaction re-exports.
    4. Update `packages/shared/src/index.ts` to re-export from the new file:
       ```typescript
       export type {
         ImportBatch,
         ImportFileError,
         ImportResponse,
         PerFileResult,
       } from './types/import.js';
       ```
       Add this block after the existing transaction type re-exports.
    5. Run `npx biome format --write packages/shared/src/types/import.ts packages/shared/src/types/index.ts packages/shared/src/index.ts`
    6. Run `npx biome check packages/shared/src/` to verify no lint issues.
    7. Run `npm run build -w packages/shared` to verify the shared package compiles.
  </action>
  <verify>
    - `npm ls csv-parse --prefix packages/backend` shows csv-parse installed
    - `npm run build -w packages/shared` succeeds
    - `npx biome check packages/shared/src/` reports no errors
  </verify>
  <done>csv-parse installed; ImportBatch, PerFileResult, ImportFileError, ImportResponse types exported from @cryptax/shared</done>
</task>
