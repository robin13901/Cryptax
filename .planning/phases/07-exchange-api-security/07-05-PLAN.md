       import type { Trade } from 'ccxt';

       export interface ExchangeAdapter {

         fetchFuturesTrades(since?: number): Promise<Trade[]>;
         testConnection(): Promise<boolean>;
       }

    2. Create `packages/backend/src/exchange/bitget-adapter.ts`:
       - Import `bitget as BitgetExchange` from 'ccxt' (ESM named export)
       - Import `type { Trade }` from 'ccxt'
       - Class `BitgetAdapter` implements `ExchangeAdapter`
       - Constructor accepts `{ apiKey: string, secret: string, password: string }` — all three required per Pitfall 1
       - `fetchSpotTrades(since?: number): Promise<Trade[]>` — calls `fetchAllTrades(undefined, since, {})`
       - `fetchFuturesTrades(since?: number): Promise<Trade[]>` — calls `fetchAllTrades(undefined, since, { type: 'swap' })` per Pitfall 2
       - `testConnection(): Promise<boolean>` — calls `this.exchange.fetchBalance()`, returns true if no error, false on catch
       - Private `fetchAllTrades(symbol, since, params)`:
         - Pagination loop: limit=100, cursor starts at `since`
         - Calls `this.exchange.fetchMyTrades(symbol, cursor, limit, params)`
         - Break conditions: batch.length === 0, batch.length < limit (last page), lastTs === cursor (no progress)
         - Cursor advancement: `cursor = lastTs + 1` (Pitfall 3: must advance past last trade)
         - Returns accumulated Trade[]

       - Test: Pagination stops when batch is empty

         - `sourceType`: the passed sourceType ('spot_tx' for spot, 'futures_tx' for futures)
         - `canonicalType`: derive from `trade.side` — 'buy' if side==='buy', 'sell' if side==='sell'. For futures, map to `futures_close_long`/`futures_close_short` based on side (simplified: treat as close since API trades are fills)
         - `symbol`: `trade.symbol` (ccxt format: 'BTC/USDT')
         - `side`: `trade.side` ('buy' | 'sell')
         - `amount`: `String(trade.amount)`
         - `price`: `String(trade.price)`
         - `fee`: `String(trade.fee?.cost ?? 0)`
         - `totalValue`: `String(trade.cost)` (price * amount)
         - `tradedAt`: `new Date(trade.timestamp).toISOString()`
         - `taxYear`: extract year from tradedAt
         - `sourceFile`: `null` (API import, no file)
         - `rawRow`: `JSON.stringify(trade)` (store ccxt Trade as raw data for audit trail)
         - `checksum`: `computeChecksum(trade)` using existing checksum module (pass trade as Record<string,string> after string conversion)
         - `importedAt`: `new Date().toISOString()`
       - Handle null/undefined fields defensively (ccxt Trade fields can be undefined)

       - `syncExchange(connectionId: number): Promise<SyncResult>` (from shared types)
       - Steps:
         a. Load exchange_connections row by id (404 if not found)
         b. Get credential master key from auth-store
         c. Decrypt credentials via `decryptCredentials(encryptedCredentials, masterKey)`
         d. Parse credentials JSON -> ExchangeCredentials
         e. Create BitgetAdapter with decrypted credentials
         f. Read `lastSyncAt` from connection (convert to Unix ms for `since` param, or undefined for first sync)
         g. Call `Promise.allSettled([adapter.fetchSpotTrades(since), adapter.fetchFuturesTrades(since)])` — partial failure pattern per CONTEXT.md

         i. For each rejected result: add to warnings array
         j. Update `last_sync_at` on exchange_connections to current ISO timestamp
         k. Trigger price enrichment and engine background (same pattern as CSV import route)
         l. Return SyncResult with counts and warnings
