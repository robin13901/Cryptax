    **packages/shared/src/decimal/money.ts:**
    ```typescript
    import Decimal from 'decimal.js'

    // Configure Decimal.js globally — max precision for crypto (up to 18 decimal places common)
    Decimal.set({ precision: 36, rounding: Decimal.ROUND_HALF_UP })

    /** Parse a TEXT DB value to Decimal — use for all monetary reads from DB */
    export function toDecimal(value: string): Decimal {
      return new Decimal(value)
    }

    /** Serialize Decimal to TEXT for DB storage — use for all monetary writes to DB */
    export function fromDecimal(value: Decimal): string {
      return value.toFixed()  // Use toFixed() to avoid scientific notation (e.g., '1e-8')
    }

    /** Zero constant — use instead of new Decimal(0) everywhere */
    export const ZERO = new Decimal(0)

    /** Add two monetary strings, return monetary string */
    export function addMoney(a: string, b: string): string {
      return fromDecimal(toDecimal(a).plus(toDecimal(b)))
    }

    /** Subtract b from a, return monetary string */
    export function subtractMoney(a: string, b: string): string {
      return fromDecimal(toDecimal(a).minus(toDecimal(b)))
    }

    /** Multiply two monetary strings, return monetary string */
    export function multiplyMoney(a: string, b: string): string {
      return fromDecimal(toDecimal(a).times(toDecimal(b)))
    }

    /** Compare: returns -1, 0, or 1 */
    export function compareMoney(a: string, b: string): number {
      return toDecimal(a).cmp(toDecimal(b))
    }

    /** Check if monetary string is zero */
    export function isZero(value: string): boolean {
      return toDecimal(value).isZero()
    }

    /** Check if monetary string is negative */
    export function isNegative(value: string): boolean {
      return toDecimal(value).isNegative()
    }

    // Re-export Decimal class for advanced usage
    export { Decimal }
    ```
