    ```typescript
    import type { Hono } from 'hono'
    export function registerHealthRoutes(app: Hono) {
      app.get('/api/health', (c) => c.json({ status: 'ok', ts: Date.now() }))
    }
    ```

    - Create app, apply logger middleware and cors on `/api/*`
    - Call `registerHealthRoutes(app)`
    - Add production static file serving (commented out for now — will be active when frontend build exists):
      Use `fileURLToPath(import.meta.url)` for proper path resolution per research pitfall #4

    - Do NOT use `serveStatic` yet — just the API routes. Static serving is a production concern.

<task type="auto">
  <name>Task 2: Wire Vite dev proxy and root dev script</name>
  <files>
    packages/frontend/vite.config.ts
    package.json
  </files>
  <action>
    **packages/frontend/vite.config.ts:**
    Add the proxy configuration to the existing Vite config:
    ```typescript
    server: {
      port: 5174,
      proxy: {
        '/api': {
          target: 'http://localhost:3001',
          changeOrigin: true,
          // NO rewrite — Hono routes are already prefixed with /api
        },
      },
    }
    ```

    **package.json (root):**
    Update the `"dev"` script to use concurrently:
    ```
    "dev": "concurrently -n backend,frontend -c blue,green \"npm run dev -w packages/backend\" \"npm run dev -w packages/frontend\""
    ```
    The `-n` flag labels the output, `-c` colors it. Backend starts first naturally (tsx is faster than Vite).

    Test the full flow: `npm run dev` from root, then verify both servers start and the proxy works.
  </action>
  <verify>
    1. `npm run dev` at root starts both backend and frontend
    2. `curl http://localhost:5174/api/health` returns `{"status":"ok","ts":...}` (proxied through Vite)
    3. Browser at http://localhost:5174 still shows the Cryptax UI
  </verify>
