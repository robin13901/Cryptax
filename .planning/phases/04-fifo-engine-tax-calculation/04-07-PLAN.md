export interface EngineRunResponse {
  success: boolean;
  /** Summary per tax year per bucket */
  summaries: Array<{
    taxYear: number;

    totalGainsEur: string;
    totalLossesEur: string;
    netEur: string;

    tradeCount: number;
  }>;
  /** FIFO engine stats */
  fifo: {
    lotsCreated: number;
    consumptions: number;
    sellsWithoutLots: number;
  };
  /** Futures engine stats */
  futures: {
    positionsRecorded: number;
  };
  /** Earn engine stats */
  earn: {
    incomeRecorded: number;
    lotsCreated: number;
  };
  /** Errors and warnings */
  errors: Array<{ transactionId: number; reason: string }>;
  /** NULL price gate failures (only populated if gate fails) */
  nullPriceErrors?: Array<{
    transactionId: number;
    symbol: string;
    tradedAt: string;
    sourceType: string;
    canonicalType: string;
  }>;
  computedAt: string;
}

1. Create `packages/backend/src/routes/engine.ts`:

Follow the same pattern as `prices.ts`:
- Module-level `let isRunning = false` concurrent guard
- Export `registerEngineRoutes(app: Hono)`

**POST /api/engine/run**:
- If isRunning: return 409 `{ error: 'Tax calculation already in progress' }`
- Set isRunning = true in try block
- Call `runTaxCalculation(db)` from the engine module
- Map the TaxCalculationResult to EngineRunResponse:
  - If result contains null price errors (check if the result has nullPriceErrors): return 422 with the errors listed
  - Otherwise: build the response with summaries, fifo stats, futures stats, earn stats, errors, computedAt
- Return 200 with EngineRunResponse
- In finally block: set isRunning = false

**GET /api/engine/status** (optional, nice-to-have):
- Return `{ isRunning }` so frontend can show a spinner

2. Update `packages/backend/src/index.ts`:
- Import `registerEngineRoutes` from `./routes/engine.js`
- Add `registerEngineRoutes(app)` after the existing route registrations

3. Create `packages/backend/src/routes/engine.test.ts`:

Test cases:
- **POST /api/engine/run returns 200 with valid data**: Set up DB with buy+sell transactions (all with eurPrice). Assert 200, response has summaries array.
- **POST /api/engine/run returns 422 when NULL prices exist**: Insert a transaction with eurPrice=null (that is taxable). Assert 422, response has nullPriceErrors.
- **POST /api/engine/run returns 409 on concurrent run**: Mock isRunning state or use a slow transaction. This test may be tricky — at minimum test that the route handler exists and returns proper structure.
- **GET /api/engine/status returns isRunning state**: Assert 200 with `{ isRunning: false }`.
