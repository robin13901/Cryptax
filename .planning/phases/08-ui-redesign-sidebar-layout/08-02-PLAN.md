**New state for sidebar collapse:**
```typescript
const [sidebarCollapsed, setSidebarCollapsed] = useState<boolean>(() => {
  try {
    return localStorage.getItem('cryptax-sidebar-collapsed') === 'true';
  } catch {
    return false;
  }
});
```

**Persist collapse state:**
```typescript
const handleToggleSidebar = () => {
  setSidebarCollapsed(prev => {
    const next = !prev;
    try { localStorage.setItem('cryptax-sidebar-collapsed', String(next)); } catch { /* ignore */ }
    return next;
  });
};
```

```tsx
// Authenticated app
return (
  <div className="app">
    {floatingLinesBg}
    <Toaster position="bottom-right" richColors />

    <Sidebar
      activeTab={activeTab}
      onTabChange={setActiveTab}
      collapsed={sidebarCollapsed}
      onToggleCollapse={handleToggleSidebar}

    <div className={`app-layout__content ${sidebarCollapsed ? 'app-layout__content--collapsed' : ''}`}>
      {/* Page title header */}
      <header className="content-header">

          {activeTab === 'dashboard' && 'Dashboard'}
          {activeTab === 'transactions' && 'Transaktionen'}
          {activeTab === 'report' && 'Steuerreport'}
          {activeTab === 'settings' && 'Einstellungen'}

      </header>

      {/* Tab Content */}
      <main className="content-main">
        <AnimatePresence mode="wait">
          {/* ... same motion.div blocks for each tab, unchanged ... */}
        </AnimatePresence>
      </main>
    </div>
  </div>
);
```
